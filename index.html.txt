    <script>
        let carrinho = [];

        function mudarQtd(idInput, valor) {
            let input = document.getElementById(idInput);
            let atual = parseInt(input.value);
            let novo = atual + valor;
            if (novo >= 1) {
                input.value = novo;
            }
        }

        function adicionarAoCarrinho(nome, preco, idQtd) {
            let qtd = parseInt(document.getElementById(idQtd).value);
            let itemExistente = carrinho.find(item => item.name === nome);
            
            if (itemExistente) {
                itemExistente.quantity += qtd;
            } else {
                carrinho.push({ name: nome, price: preco, quantity: qtd });
            }
            
            atualizarBadge();
            alert(qtd + "x " + nome + " adicionado(s) ao carrinho!");
        }

        function atualizarBadge() {
            let totalItens = carrinho.reduce((sum, item) => sum + item.quantity, 0);
            document.getElementById('badge-carrinho').innerText = totalItens;
        }

        function abrirCarrinho() {
            renderizarCarrinho();
            document.getElementById('modal-carrinho').style.display = 'block';
        }

        function fecharCarrinho() {
            document.getElementById('modal-carrinho').style.display = 'none';
        }

        function renderizarCarrinho() {
            let container = document.getElementById('itens-carrinho-container');
            if (carrinho.length === 0) {
                container.innerHTML = '<p style="text-align: center; color: #7f8c8d; padding: 20px;">Seu carrinho está vazio.</p>';
            } else {
                container.innerHTML = carrinho.map((item, index) => `
                    <div class="item-carrinho">
                        <span>${item.quantity}x ${item.name}</span>
                        <span>R$ ${(item.price * item.quantity).toFixed(2).replace('.', ',')}</span>
                        <button onclick="removerItem(${index})" style="background:none; border:none; color:#e74c3c; cursor:pointer; font-weight:bold;">✕</button>
                    </div>
                `).join('');
            }
            atualizarTotais();
        }

        function removerItem(index) {
            carrinho.splice(index, 1);
            atualizarBadge();
            renderizarCarrinho();
        }

        function atualizarTotais() {
            let subtotal = carrinho.reduce((sum, item) => sum + (item.price * item.quantity), 0);
            let diasSelect = document.getElementById('dias-aluguel');
            let dias = diasSelect ? parseInt(diasSelect.value) : 1;
            
            // Cada dia além do 1º dia adiciona R$ 100
            let taxaDias = (dias - 1) * 100;
            let totalGeral = subtotal + taxaDias;

            document.getElementById('subtotal-produtos').innerText = 'R$ ' + subtotal.toFixed(2).replace('.', ',');
            document.getElementById('taxa-dias').innerText = 'R$ ' + taxaDias.toFixed(2).replace('.', ',');
            document.getElementById('total-geral').innerText = 'R$ ' + totalGeral.toFixed(2).replace('.', ',');
        }

        function filtrarMusicasExemplo() {
            let termo = document.getElementById('campoBuscaMusica').value.trim();
            let divRes = document.getElementById('resultadoBusca');
            if(termo.length > 2) {
                divRes.innerHTML = `🔍 Buscando por "${termo}" no catálogo de milhares de músicas... Disponível!`;
            } else {
                divRes.innerHTML = '';
            }
        }

        function gerarPagamentoPix() {
            let nome = document.getElementById('cliente-nome').value;
            let tel = document.getElementById('cliente-tel').value;
            let data = document.getElementById('cliente-data').value;
            let endereco = document.getElementById('cliente-endereco').value;

            if (!nome || !tel || !data || !endereco) {
                alert('Por favor, preencha todos os campos do formulário (Nome, Telefone, Data e Endereço) antes de prosseguir para o PIX.');
                return;
            }

            if (carrinho.length === 0) {
                alert('O seu carrinho está vazio.');
                return;
            }

            // Ocultar formulário e mostrar painel PIX
            document.getElementById('form-checkout').style.display = 'none';
            document.getElementById('area-pix').style.display = 'block';
            iniciarContagemRegressiva(14, 59);
        }

        function copiarChavePix() {
            let textoCopia = document.getElementById('codigo-pix-copia').innerText;
            navigator.clipboard.writeText(textoCopia).then(() => {
                alert('Código PIX Copia e Cola copiado com sucesso!');
            });
        }

        function iniciarContagemRegressiva(minutos, segundos) {
            let tempo = minutos * 60 + segundos;
            let timerSpan = document.getElementById('timer-pix');
            
            let intervalo = setInterval(function () {
                let m = Math.floor(tempo / 60);
                let s = tempo % 60;
                m = m < 10 ? '0' + m : m;
                s = s < 10 ? '0' + s : s;
                timerSpan.innerText = m + ':' + s;
                
                if (--tempo < 0) {
                    clearInterval(intervalo);
                    timerSpan.innerText = 'Expirado';
                }
            }, 1000);
        }
